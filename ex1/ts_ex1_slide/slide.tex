\section{Mô hình ARIMA}
% ============================================================

\begin{frame}{Mô hình ARIMA}
    \begin{block}{ARIMA(p, d, q)}
        \begin{itemize}
            \item \highlight{p}: Bậc Tự hồi quy (AR)
            \item \highlight{d}: Bậc sai phân (Integration)
            \item \highlight{q}: Bậc Trung bình trượt (MA)
        \end{itemize}
    \end{block}

    \vspace{0.4cm}
    \begin{alertblock}{Đặc điểm}
        \begin{itemize}
            \item Mô hình \highlight{đơn biến} -- mỗi chuỗi được mô hình hóa \highlight{độc lập}
            \item Không nắm bắt tương quan giữa các biến
            \item Grid search trên (p, d, q) để tối thiểu AIC
        \end{itemize}
    \end{alertblock}
\end{frame}

\section{Mô hình VAR}
% ============================================================

\begin{frame}{Mô hình VAR}
    \begin{block}{VAR(k) -- Vector Autoregression}
        \begin{itemize}
            \item Mô hình \highlight{đa biến} -- tất cả biến được mô hình hóa \highlight{cùng lúc}
            \item Nắm bắt \highlight{tương quan chéo} giữa các biến
            \item Mỗi biến phụ thuộc vào giá trị quá khứ của \highlight{tất cả} biến
        \end{itemize}
    \end{block}

    \vspace{0.3cm}
    \begin{equation*}
        \mathbf{Y}_t = \mathbf{c} + \mathbf{A}_1 \mathbf{Y}_{t-1} + \cdots + \mathbf{A}_k \mathbf{Y}_{t-k} + \boldsymbol{\varepsilon}_t
    \end{equation*}
    với $\mathbf{Y}_t = [CO, C6H6, NO_2, T]^T$
\end{frame}

\begin{frame}{Chọn số bậc trễ tối ưu}
    \begin{center}
        \begin{tabular}{|r|r|}
            \hline
            \textbf{Bậc (lag)} & \textbf{AIC} \\ \hline
            \red{1} & \red{7.18} \\ \hline
            2 & 7.26 \\ \hline
            3 & 7.23 \\ \hline
            4 & 7.27 \\ \hline
            5 & 7.34 \\ \hline
            ... & ... \\ \hline
            15 & 7.77 \\ \hline
        \end{tabular}
    \end{center}

    \vspace{0.4cm}
    \begin{block}{Kết luận}
        Số bậc trễ tối ưu: \highlight{k = 1} (AIC = 7.18)

\end{frame}
